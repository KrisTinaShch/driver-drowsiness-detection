# Dependencies of the driver drowsiness detector.
#
# Deliberately unpinned. The project started on macOS/Intel with Python 3.9;
# those pins have no prebuilt wheels on Windows with Python 3.10, so versions
# are resolved per machine instead.

# --- tables and plots ---
numpy
pandas
matplotlib
scikit-learn

# --- training the network ---
torch
torchvision

# --- images and webcam video ---
# install the contrib build only: plain opencv-python ships the very same cv2
# module and the two conflict
opencv-contrib-python

# --- locating the face and the eyes in a frame ---
mediapipe

# --- rendering labels (OpenCV cannot draw Cyrillic) ---
pillow

# --- fetching the dataset from Kaggle (works without a token) ---
kagglehub

# --- audible alarm ---
sounddevice

# --- experiment tracking (logs runs into a local SQLite file) ---
mlflow

# --- tests ---
pytest

# --- notebook environment ---
jupyterlab
